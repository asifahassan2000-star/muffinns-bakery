/* Entrance and scroll motion recreations; no Framer runtime required. */
(function(){
const reduced=window.matchMedia('(prefers-reduced-motion: reduce)').matches;
const reveal=Array.from(document.querySelectorAll('.reveal'));
if(!reduced && 'IntersectionObserver' in window){
 document.documentElement.classList.add('motion-ready');
 const observer=new IntersectionObserver(entries=>{entries.forEach(entry=>{
  if(entry.isIntersecting){entry.target.classList.add('is-visible');observer.unobserve(entry.target);}
 });},{threshold:.15});
 reveal.forEach(el=>{
  el.style.setProperty('--delay',(Number(el.dataset.delay)||0)+'s');
  if(el.dataset.fromY)el.style.setProperty('--from-y',el.dataset.fromY+'px');
  if(el.dataset.fromX)el.style.setProperty('--from-x',el.dataset.fromX+'px');
  if(el.dataset.mount==='true')requestAnimationFrame(()=>requestAnimationFrame(()=>el.classList.add('is-visible')));
  else observer.observe(el);
 });
}
const nav=document.querySelector('.nav');
const toggle=document.querySelector('.menu-toggle');
function closeMenu(){nav.classList.remove('open');toggle.setAttribute('aria-expanded','false');toggle.setAttribute('aria-label','Open navigation');}
toggle.addEventListener('click',()=>{const open=nav.classList.toggle('open');toggle.setAttribute('aria-expanded',String(open));toggle.setAttribute('aria-label',open?'Close navigation':'Open navigation');});
document.addEventListener('keydown',e=>{if(e.key==='Escape'){closeMenu();}});
document.querySelectorAll('.nav-links a').forEach(a=>a.addEventListener('click',closeMenu));
window.matchMedia('(min-width:1200px)').addEventListener('change',e=>{if(e.matches)closeMenu();});
/* Set each form's data-endpoint to your newsletter handler URL.
   Expected: HTTPS endpoint accepting JSON { email: "..." } with CORS enabled.
   No subscription success is shown unless the server confirms success. */
document.querySelectorAll('[data-newsletter]').forEach(form=>form.addEventListener('submit',async e=>{
 e.preventDefault(); if(!form.reportValidity())return;
 const status=form.parentElement.querySelector('[role="status"]');
 const endpoint=form.dataset.endpoint;
 if(!endpoint){status.textContent='Signup is not connected yet. Please contact the restaurant directly.';return;}
 const button=form.querySelector('button');button.disabled=true;status.textContent='Submitting…';
 try{const response=await fetch(endpoint,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({email:form.querySelector('[type="email"]').value})});if(!response.ok)throw new Error('Request failed');status.textContent='Thanks! You’re on the list.';form.reset();}
 catch(error){status.textContent='We couldn’t complete signup. Please try again later.';}
 finally{button.disabled=false;}
}));
})();
