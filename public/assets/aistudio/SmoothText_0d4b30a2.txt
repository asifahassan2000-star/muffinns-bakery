import * as React from "react"
import { motion, useReducedMotion } from "framer-motion"
import { addPropertyControls, ControlType } from "framer"

interface Props {
 text: string; color: string; font: any; delay: number; stagger: number
 duration: number; align: "left" | "center" | "right"; style?: React.CSSProperties
}
/** @framerSupportedLayoutWidth any
 * @framerSupportedLayoutHeight auto */
export default function SmoothText(props: Props) {
 const { text = "Elegance plated perfectly", color = "#222222", font, delay = 0,
 stagger = .07, duration = .9, align = "left", style } = props
 const reduced = useReducedMotion()
 const words = text.trim().split(/\s+/)
 return <motion.div role="heading" aria-level={2} aria-label={text}
 initial="hidden" whileInView="visible" viewport={{ once: true, amount: .25 }}
 style={{ width:"100%", fontFamily:'"DM Serif Text", serif', fontSize:48,
 lineHeight:1.2, fontWeight:400, ...font, color, textAlign:align, ...style }}>
 {words.map((word, i) => <React.Fragment key={i}>
 <span aria-hidden="true" style={{display:"inline-block",overflow:"hidden",verticalAlign:"top",paddingBottom:".14em",marginBottom:"-.14em"}}>
 <motion.span style={{display:"inline-block"}}
 variants={{hidden:{opacity:reduced?1:0,y:reduced?0:"105%"},visible:{opacity:1,y:0}}}
 transition={{duration:reduced?0:duration,delay:reduced?0:delay+i*stagger,ease:[.22,1,.36,1]}}>
 {word}</motion.span></span>{i<words.length-1?" ":null}
 </React.Fragment>)}</motion.div>
}
addPropertyControls(SmoothText,{
 text:{type:ControlType.String,title:"Text",defaultValue:"Elegance plated perfectly",displayTextArea:true},
 font:{type:ControlType.Font,title:"Font",controls:"extended",defaultFontType:"serif"},
 color:{type:ControlType.Color,title:"Color",defaultValue:"#222222"},
 align:{type:ControlType.Enum,title:"Align",options:["left","center","right"],defaultValue:"left"},
 duration:{type:ControlType.Number,title:"Duration",defaultValue:.9,min:.1,max:3,step:.05},
 delay:{type:ControlType.Number,title:"Delay",defaultValue:0,min:0,max:3,step:.05},
 stagger:{type:ControlType.Number,title:"Stagger",defaultValue:.07,min:0,max:.3,step:.01}
})
